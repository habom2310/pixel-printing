API

One call per object

const print = pixelPrint(target, {
  effect: "scan",   // "scan" | "dust" | "layers"
  depth: "2d",      // "2d" | "3d"
  spread: 0.3,      // how far apart parts start
  color: "#5eead4", // print head and tint
  start: "visible", // or "now"
});
await print.finished; // or print.finish(), print.destroy(), print.setSpeed(2)
